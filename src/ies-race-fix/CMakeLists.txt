cmake_minimum_required(VERSION 3.20)
project(IesRaceFix LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OSX_ARCHITECTURES "arm64")
set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0")
set(RED4EXT_SDK_DIR "" CACHE PATH "RED4ext.SDK-macos checkout with patches/RED4ext.SDK.diff applied")
if(NOT RED4EXT_SDK_DIR)
    message(FATAL_ERROR "Pass -DRED4EXT_SDK_DIR=<path to RED4ext.SDK-macos>")
endif()
add_library(IesRaceFix SHARED Main.cpp)
set_target_properties(IesRaceFix PROPERTIES PREFIX "" SUFFIX ".dylib")
target_include_directories(IesRaceFix PRIVATE "${RED4EXT_SDK_DIR}/include")
target_compile_definitions(IesRaceFix PRIVATE RED4EXT_PLATFORM_MACOS)
target_compile_options(IesRaceFix PRIVATE -fvisibility=hidden -Wno-deprecated-declarations)
target_link_options(IesRaceFix PRIVATE -undefined dynamic_lookup)
